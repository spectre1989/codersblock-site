{{/* Usage: {{}} or: {{}} (resize="" disables resizing) By default images wider than 1000px are scaled down; smaller ones are left alone. */}} {{ $asset_path := "" }} {{ $resize := "1000x" }} {{ $explicit := false }} {{ if .IsNamedParams }} {{ $asset_path = .Get "src" }} {{ range $k, $v := .Params }} {{ if eq $k "resize" }}{{ $resize = $v }}{{ $explicit = true }}{{ end }} {{ end }} {{ else }} {{ $asset_path = .Get 0 }} {{ end }} {{ $img := resources.Get $asset_path }} {{ if not $img }}{{ errorf "asset_img: %q not found (%s)" $asset_path .Position }}{{ end }} {{ if and $resize (or $explicit (gt $img.Width 1000)) }} {{ $img = $img.Resize $resize }} {{ end }}
